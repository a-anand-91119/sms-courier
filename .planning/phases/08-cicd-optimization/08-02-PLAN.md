Replace the `generate_changelog` lane implementation:

Current (lines 59-82):
```ruby
desc "Generate changelog from git commits"
lane :generate_changelog do |options|
  version_code = options[:version_code] || ENV["VERSION_CODE"] || "1"
  ...
  changelog = changelog_from_git_commits(...)
  ...
end
```

Replace with:
```ruby
desc "Read changelog from CHANGELOG.md for Play Store upload"
lane :generate_changelog do |options|
  version_code = options[:version_code] || ENV["VERSION_CODE"] || "1"

  if version_code.to_s.strip.empty?
    UI.user_error!("version_code cannot be empty")
  end

  # Read the [Unreleased] section from CHANGELOG.md
  changelog = read_changelog(
    changelog_path: './CHANGELOG.md',
    section_identifier: '[Unreleased]',
    excluded_markdown_elements: ['###']  # Strip subsection headers for Play Store
  )

  # Fallback if [Unreleased] is empty - read most recent version
  if changelog.strip.empty?
    UI.important("No unreleased changes, reading most recent version section")
    changelog = read_changelog(
      changelog_path: './CHANGELOG.md',
      section_identifier: '[0',  # Matches [0.x.x] sections
      excluded_markdown_elements: ['###']
    )
  end

  # Write to Play Store changelog format
  changelogs_dir = "fastlane/metadata/android/en-US/changelogs"
  FileUtils.mkdir_p(changelogs_dir) unless Dir.exist?(changelogs_dir)

  changelog_path = "#{changelogs_dir}/#{version_code}.txt"
  File.write(changelog_path, changelog)

  UI.success("Changelog written to #{changelog_path}")
end
```

Key changes:
- Uses `read_changelog` from plugin instead of `changelog_from_git_commits`
- Reads curated content from CHANGELOG.md
- Strips markdown headers for cleaner Play Store notes
- Has fallback if Unreleased is empty
